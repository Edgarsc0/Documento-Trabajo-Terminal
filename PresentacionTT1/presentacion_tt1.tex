% ============================================================
%  7. Modelo formal
% ============================================================
\begin{frame}{Un juego es un sistema de transición de estados}
\framesubtitle{Modelo formal}
\begin{banda}[vinooscuro]
\centering\Large $\Gamma=\langle R,\ S,\ s_0,\ \{A_i\},\ \{L_i\},\ T,\ F,\ \{G_i\}\rangle$
\end{banda}
\vspace{2pt}
\begin{columns}[T]
\column{.48\textwidth}
\small
\begin{itemize}\setlength\itemsep{2pt}
\item[$R$] Roles o jugadores.
\item[$S$] Estados, como conjuntos de hechos (\emph{fluents}).
\item[$s_0$] Estado inicial.
\item[$A_i$] Acciones de cada rol.
\end{itemize}
\column{.48\textwidth}
\small
\begin{itemize}\setlength\itemsep{2pt}
\item[$L_i$] Legalidad: $L_i:S\to\mathcal{P}(A_i)$.
\item[$T$] Transición determinista: $T:S\times\mathbf{A}\to S$.
\item[$F$] Estados terminales.
\item[$G_i$] Utilidad: $G_i:S\to[0,100]$.
\end{itemize}
\end{columns}
\vspace{4pt}
\begin{tarjeta}[azulp]
\footnotesize
\textbf{Interfaz hacia el motor:} estado inicial, acciones legales, aplicar acción, \emph{¿es terminal?} y utilidad. Un juego bien formado cumple \emph{jugabilidad} (todo rol tiene una acción legal en estados no terminales) y \emph{terminación} garantizada. Los turnos alternados usan la acción nula \texttt{noop}.
\end{tarjeta}
\fuente{Cap.~3, \S3.4 \cite{Genesereth2005GGP, Love2008GDL, GeneserethThielscher2014}.}
\end{frame}

\begin{frame}[fragile]{Reglas externas: GDL, subconjunto o DSL}
\framesubtitle{Lenguaje de descripción}
\begin{columns}[T]
\column{.47\textwidth}
{\scriptsize\bfseries\color{vino} EJEMPLO EN GDL: GATO}\par\vspace{2pt}
\begin{lstlisting}
; legalidad
(<= (legal ?p (mark ?r ?c))
    (true (control ?p))
    (true (cell ?r ?c b)))

; alternancia de turno
(<= (next (control o))
    (true (control x)))
(<= (next (control x))
    (true (control o)))
\end{lstlisting}
{\scriptsize Las reglas se \emph{declaran}: el motor deduce las consecuencias.\par}
\column{.51\textwidth}
\footnotesize
\renewcommand{\arraystretch}{1.25}
\begin{tabularx}{\linewidth}{@{}>{\bfseries\raggedright\arraybackslash}p{2.0cm}>{\raggedright\arraybackslash}X@{}}
\toprule
GDL completo & Base académica y semántica lógica general; puede resultar pesado para un prototipo acotado. \\
\midrule
Subconjunto & Puede simplificar la implementación; exige delimitar y justificar qué se admite. \\
\midrule
DSL propio & Se concentra en lo que el proyecto necesita y puede ser más legible (como Ludii); debe justificarse formalmente. \\
\bottomrule
\end{tabularx}
\end{columns}
\fuente{Cap.~3, \S3.5--\S3.6 \cite{Love2008GDL, Soemers2024Ludii}.}
\end{frame}

% ---------- B. Algoritmos ----------
\begin{frame}{Minimax, alfa-beta y MCTS con UCT}
\framesubtitle{Respaldo B}
\footnotesize
\begin{columns}[T]
\column{.485\textwidth}
\begin{tarjeta}[vino]
{\bfseries Minimax y poda alfa-beta}\\[2pt]
$V(s)=\begin{cases}G_1(s) & s\in F\\ \max_{a\in L_1(s)}V(T(s,(a,\text{noop}))) & \text{turno de }r_1\\ \min_{a\in L_2(s)}V(T(s,(\text{noop},a))) & \text{turno de }r_2\end{cases}$\\[3pt]
Alfa-beta no cambia la decisión: poda ramas que no pueden mejorar el resultado conocido. Con profundidad limitada se requiere una evaluación general.
\end{tarjeta}
\column{.485\textwidth}
\begin{tarjeta}[azulp]
{\bfseries MCTS y selección UCT}\\[2pt]
Ciclo: selección, expansión, simulación y retropropagación.\\[2pt]
$v^{*}=\arg\max_{v'}\left[\dfrac{Q_{p(v)}(v')}{N(v')}+C\sqrt{\dfrac{\ln N(v)}{N(v')}}\right]$\\[3pt]
Primero se seleccionan hijos sin visitas. $Q_{p(v)}$ acumula recompensas normalizadas del jugador activo.\\[2pt]
Equilibra explotación y exploración sin heurísticas propias de un juego.
\end{tarjeta}
\end{columns}
\fuente{Cap.~3, \S3.9--\S3.13 \cite{Browne2012MCTS, Kocsis2006UCT, RussellNorvig2010}.}
\end{frame}
